cmake_minimum_required(VERSION 3.16)
project(triton_agent1 VERSION 0.2.0 LANGUAGES C)
include(CTest)
add_executable(triton-agent1 csrc/triton_agent1_main.c)
target_include_directories(triton-agent1 PRIVATE include)
set_target_properties(triton-agent1 PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
option(TRITON_SANITIZE "Enable address/undefined sanitizers" OFF)
if(MSVC)
  target_compile_options(triton-agent1 PRIVATE /W4)
else()
  target_compile_options(triton-agent1 PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(TRITON_SANITIZE)
    target_compile_options(triton-agent1 PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(triton-agent1 PRIVATE -fsanitize=address,undefined)
  endif()
endif()
configure_file(examples/program.txt program.txt COPYONLY)
configure_file(examples/invalid.txt invalid.txt COPYONLY)
add_test(NAME smoke-jovial COMMAND triton-agent1 jovial program.txt)
add_test(NAME smoke-cms2 COMMAND triton-agent1 cms2 program.txt)
add_test(NAME smoke-tacpol COMMAND triton-agent1 tacpol program.txt)
add_test(NAME rejects-invalid COMMAND triton-agent1 jovial invalid.txt)
set_tests_properties(rejects-invalid PROPERTIES WILL_FAIL TRUE)
install(TARGETS triton-agent1 RUNTIME DESTINATION bin)
install(FILES include/triton_agent1.h DESTINATION include)
install(FILES LICENSE GOVERNANCE.md DESTINATION share/triton-agent1)

if(MSVC)
  set_property(TARGET triton-agent1 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
